"""
Jarvis 4.1 - Enhanced Laptop Assistant
Features added:
- Reused TTS engine with atexit cleanup to reduce pyttsx3 warnings
- Usage tracking (counts) for recognized commands
- Auto-learning / suggestion: optionally save frequently used commands as "routines"
- Routines: named sequences of commands saved in brain.json and runnable by voice ("run routine morning")
- More robust listen() with timeouts & graceful exception handling
- Config file (brain.json) structure: {"commands": {cmd: action}, "usage": {cmd: count}, "routines": {name: [cmd1, cmd2]}}

How to use:
1. Install required packages:
   pip install SpeechRecognition pyttsx3 pyaudio pywhatkit wikipedia keyboard pyautogui
   (If PyAudio fails on Windows -> install matching .whl from Gohlke site)
2. Put this file in your Jarvis folder and run:
   python Jarvis4_enhanced.py
3. Speak commands like:
   - "open chrome"
   - "search google python tutorials"
   - "play despacito"
   - "wikipedia python"
   - "screenshot"
   - "type hello world"
   - "run routine morning"
   - "create routine morning" (then follow prompts in console)
   - "exit"

Note: when asked to teach a missing command the script will ask in the console (text input). If you prefer voice-only teaching, that can be added later.

"""

BRAIN_FILE = "brain.json"
SPEAK_LOCK = threading.Lock()

# ------------------ UTIL: Brain load/save ------------------

def load_brain():
    try:
        with open(BRAIN_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
    except FileNotFoundError:
        data = {"commands": {}, "usage": {}, "routines": {}}
    # ensure keys
    data.setdefault("commands", {})
    data.setdefault("usage", {})
    data.setdefault("routines", {})
    return data


def save_brain(data):
    with open(BRAIN_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)

# ------------------ TTS Engine (single instance) ------------------

engine = pyttsx3.init()
# optional: tweak voice/rate here if you want
try:
    engine.setProperty('rate', 170)
except Exception:
    pass


def speak(text, block=True):
    """Speak text. If block is False, speech will be queued but function returns immediately."""
    # Protect pyttsx3 from concurrent calls
    def _speak():
        with SPEAK_LOCK:
            try:
                engine.say(text)
                engine.runAndWait()
            except Exception:
                # pyttsx3 sometimes throws on shutdown - ignore
                pass

    if block:
        _speak()
    else:
        t = threading.Thread(target=_speak, daemon=True)
        t.start()


# Ensure engine is properly stopped at exit to reduce warnings
def _cleanup_tts():
    try:
        engine.stop()
    except Exception:
        pass

atexit.register(_cleanup_tts)

# ------------------ Recognizer / Listening ------------------

recognizer = sr.Recognizer()


def listen(timeout=5, phrase_time_limit=6):
    """Listen and return recognized lowercased text or empty string on timeout/unknown."""
    with sr.Microphone() as source:
        # Adjust for ambient noise. Small value to react faster.
        recognizer.adjust_for_ambient_noise(source, duration=0.6)
        print("Listening...")
        try:
            audio = recognizer.listen(source, timeout=timeout, phrase_time_limit=phrase_time_limit)
            text = recognizer.recognize_google(audio)
            print("User:", text)
            return text.lower()
        except sr.WaitTimeoutError:
            return ""
        except sr.UnknownValueError:
            speak("I could not understand. Please repeat.")
            return ""
        except sr.RequestError:
            speak("Speech API error. Check your internet connection.")
            return ""
        except Exception as e:
            # catch other audio backend issues
            print("Listen error:", e)
            return ""

# ------------------ COMMAND EXECUTION & LOGIC ------------------

brain = load_brain()


def increment_usage(cmd):
    brain.setdefault("usage", {})
    brain["usage"][cmd] = brain["usage"].get(cmd, 0) + 1
    # persist usage quickly so frequent-command suggestions survive crashes
    save_brain(brain)


def run_routine(routine_name):
    routines = brain.get("routines", {})
    if routine_name in routines:
        steps = routines[routine_name]
        speak(f"Running routine {routine_name}")
        for step in steps:
            speak(f"Now: {step}", block=False)
            execute(step)
            time.sleep(1)
    else:
        speak(f"I do not have a routine named {routine_name}")


def create_routine_via_console(routine_name):
    print(f"Creating routine: {routine_name}. Enter voice commands line by line. Empty line to finish.")
    steps = []
    while True:
        cmd = input("Command: ").strip().lower()
        if cmd == "":
            break
        steps.append(cmd)
    if steps:
        brain.setdefault("routines", {})
        brain["routines"][routine_name] = steps
        save_brain(brain)
        speak(f"Routine {routine_name} saved. It has {len(steps)} steps.")
    else:
        print("No steps provided. Routine cancelled.")


# Core execute function

def execute(command):
    if not command:
        return
    # increment usage for analytics
    increment_usage(command)

    # exit
    if command == "exit" or command == "quit":
        speak("Shutting down. Goodbye!")
        return "exit"

    # run routine
    if command.startswith("run routine "):
        name = command.replace("run routine ", "").strip()
        run_routine(name)
        return
    if command.startswith("create routine "):
        name = command.replace("create routine ", "").strip()
        # small console interaction to capture steps
        speak(f"Creating routine {name}. Check console to enter steps.")
        create_routine_via_console(name)
        return

    # ---- open apps mapping ----
    apps = {
        "chrome": "start chrome",
        "vs code": "code",
        "vscode": "code",
        "notepad": "notepad",
        "calculator": "calc",
        "spotify": "start spotify",
        "youtube": "start chrome https://www.youtube.com"
    }
    for key, action in apps.items():
        if f"open {key}" in command:
            speak(f"Opening {key}")
            os.system(action)
            return

    # ---- google search ----
    if command.startswith("search google"):
        query = command.replace("search google", "").strip()
        if query:
            speak(f"Searching Google for {query}")
            webbrowser.open(f"https://www.google.com/search?q={query}")
        else:
            speak("Please tell me what to search.")
        return

    # ---- play on YouTube ----
    if command.startswith("play "):
        song = command.replace("play", "").strip()
        if song:
            speak(f"Playing {song} on YouTube")
            try:
                pywhatkit.playonyt(song)
            except Exception:
                webbrowser.open(f"https://www.youtube.com/results?search_query={song}")
        return

    # ---- wikipedia ----
    if command.startswith("wikipedia"):
        topic = command.replace("wikipedia", "").strip()
        if not topic:
            speak("Which topic on Wikipedia?")
            topic = listen()
        if topic:
            try:
                summary = wikipedia.summary(topic, sentences=2)
                speak(summary)
            except Exception:
                speak("Couldn't find that on Wikipedia.")
        return

    # ---- system controls ----
    if "shutdown" in command:
        speak("Shutting down PC in 5 seconds")
        os.system("shutdown /s /t 5")
        return
    if "restart" in command:
        speak("Restarting PC in 5 seconds")
        os.system("shutdown /r /t 5")
        return
    if "lock" in command:
        speak("Locking PC")
        os.system("rundll32.exe user32.dll,LockWorkStation")
        return

    # ---- screenshot ----
    if "screenshot" in command:
        try:
            filename = f"screenshot_{int(time.time())}.png"
            img = pyautogui.screenshot()
            img.save(filename)
            speak(f"Screenshot saved as {filename}")
        except Exception:
            speak("Failed to take screenshot")
        return

    # ---- keyboard automation ----
    if command.startswith("press "):
        key = command.replace("press ", "").strip()
        try:
            keyboard.press_and_release(key)
            speak(f"Pressed {key}")
        except Exception:
            speak(f"Could not press {key}")
        return
    if command.startswith("type "):
        text = command.replace("type ", "").strip()
        try:
            keyboard.write(text)
            speak(f"Typed {text}")
        except Exception:
            speak("Typing failed")
        return

    # ---- check brain custom commands ----
    custom = brain.get("commands", {})
    if command in custom:
        action = custom[command]
        speak(f"Executing saved command for {command}")
        try:
            os.system(action)
        except Exception:
            speak("Failed to run saved command")
        return

    # ---- unknown command: offer to teach (console fallback) ----
    speak("I don't know this command yet. Would you like to teach me? Say yes to teach.")
    reply = listen()
    if "yes" in reply or "yeah" in reply or "yup" in reply:
        speak("Please type the system command in the console now.")
        action = input(f"Enter system command to run for '{command}': ")
        if action:
            brain.setdefault("commands", {})
            brain["commands"][command] = action
            save_brain(brain)
            speak("Saved. I will remember this command next time.")
    else:
        speak("Okay, I won't save it.")

# ------------------ Startup suggestions ------------------

def top_commands_suggestion(limit=3):
    usage = brain.get("usage", {})
    if not usage:
        return
    # sort by count
    sorted_cmds = sorted(usage.items(), key=lambda x: x[1], reverse=True)
    top = [c for c, _ in sorted_cmds[:limit]]
    if top:
        speak("You often use these commands:")
        for t in top:
            speak(t, block=False)
        time.sleep(1)
        speak("You can ask me to create a routine from them. Say create routine followed by a name.")

# ------------------ Main loop ------------------

def run_jarvis():
    speak("Jarvis activated. I'm ready.")
    # Give quick suggestion of top commands (non-blocking)
    threading.Thread(target=top_commands_suggestion, daemon=True).start()
    try:
        while True:
            cmd = listen()
            if not cmd:
                continue
            res = execute(cmd)
            if res == "exit":
                break
    except KeyboardInterrupt:
        print("Keyboard interrupt - exiting")
    finally:
        save_brain(brain)
        _cleanup_tts()

# ------------------ Entry ------------------

if __name__ == "__main__":
    run_jarvis()

# Voice-only teaching: teach a new command mapping via voice
# Usage: "teach command <phrase> as <system command>"

def teach_command_via_voice(command_text):
    # Parse basic pattern: 'teach command X as Y'
    # fallback: ask follow-up questions
    if "teach command" in command_text and " as " in command_text:
        try:
            _, rest = command_text.split("teach command", 1)
            phrase_part, action_part = rest.split(" as ", 1)
            phrase = phrase_part.strip()
            action = action_part.strip()
            if phrase and action:
                brain.setdefault("commands", {})
                brain["commands"][phrase] = action
                save_brain(brain)
                speak(f"Saved command for {phrase}")
                return True
        except Exception:
            pass
    # fallback interactive voice: ask for phrase then ask for action
    speak("Please speak the phrase I should listen for")
    phrase = listen()
    if not phrase:
        speak("No phrase detected. Cancelled.")
        return False
    speak("Now speak the system command I should run when you say that phrase")
    action = listen(10, 8)
    if not action:
        speak("No action detected. Cancelled.")
        return False
    brain.setdefault("commands", {})
    brain["commands"][phrase] = action
    save_brain(brain)
    speak(f"Learned: when you say {phrase} I will run {action}")
    return True

# Create routine via voice-only: 'create routine <name>' then say steps one-by-one and 'done'

def create_routine_via_voice(routine_name):
    speak(f"Creating routine {routine_name}. Say each voice command you want to add. Say done when finished.")
    steps = []
    while True:
        step = listen(8, 6)
        if not step:
            speak("I didn't hear anything. Say done to finish or continue to add steps.")
            cont = listen(4,4)
            if "done" in cont:
                break
            else:
                continue
        if step.strip() in ("done", "finish", "that's it"):
            break
        steps.append(step.strip())
        speak(f"Added step: {step}", block=False)
    if steps:
        brain.setdefault("routines", {})
        brain["routines"][routine_name] = steps
        save_brain(brain)
        speak(f"Routine {routine_name} saved with {len(steps)} steps")
        return True
    else:
        speak("No steps added. Routine cancelled")
        return False

# Schedule routine: support "schedule routine <name> at HH:MM [daily]"

def schedule_routine_via_voice(command_text):
    # simple parsing
    parts = command_text.split()
    if "schedule routine" in command_text:
        try:
            # extract routine name between 'schedule routine' and 'at'
            after = command_text.split("schedule routine",1)[1].strip()
            if " at " in after:
                name_part, time_part = after.split(" at ",1)
                name = name_part.strip()
                # time may include 'daily'
                if " daily" in time_part:
                    time_str = time_part.replace(" daily","",1).strip()
                    repeat = "daily"
                else:
                    time_str = time_part.strip()
                    repeat = "once"
                # normalize time
                try:
                    t = datetime.strptime(time_str, "%H:%M").time()
                except Exception:
                    # try 12h format
                    try:
                        t = datetime.strptime(time_str, "%I:%M %p").time()
                    except Exception:
                        speak("Could not parse time. Use HH:MM in 24 hour or HH:MM AM/PM formats.")
                        return False
                sched = {"routine": name, "time": t.strftime("%H:%M"), "repeat": repeat}
                brain.setdefault("schedules", [])
                brain["schedules"].append(sched)
                save_brain(brain)
                speak(f"Scheduled routine {name} at {sched['time']} {repeat}")
                return True
        except Exception as e:
            print("Schedule parse error", e)
    speak("Could not schedule. Try saying: schedule routine morning at 07:30 daily")
    return False

# Bind hotkey to routine: 'bind routine <name> to ctrl alt m' (spoken keys separated by spaces)

def bind_hotkey_via_voice(command_text):
    if "bind routine" in command_text and " to " in command_text:
        try:
            after = command_text.split("bind routine",1)[1].strip()
            name_part, keys_part = after.split(" to ",1)
            name = name_part.strip()
            keys = keys_part.strip().replace(' ', '+')
            # register hotkey
            def _runner():
                run_routine(name)
            try:
                keyboard.add_hotkey(keys, _runner)
                brain.setdefault("hotkeys", {})
                brain["hotkeys"][keys] = name
                save_brain(brain)
                speak(f"Bound routine {name} to hotkey {keys}")
                return True
            except Exception as e:
                print("Hotkey bind error", e)
        except Exception as e:
            print("Bind parse error", e)
    speak("Could not bind hotkey. Try: bind routine morning to ctrl alt m")
    return False

# ------------------ Scheduler background thread ------------------

def scheduler_loop():
    while True:
        now = datetime.now()
        current_time = now.strftime("%H:%M")
        schedules = brain.get("schedules", [])
        for sched in list(schedules):
            sched_time = sched.get("time")
            routine = sched.get("routine")
            repeat = sched.get("repeat", "once")
            if sched_time == current_time:
                # run routine in separate thread
                threading.Thread(target=run_routine, args=(routine,), daemon=True).start()
                if repeat != "daily":
                    # remove one-time schedule
                    try:
                        brain["schedules"].remove(sched)
                        save_brain(brain)
                    except Exception:
                        pass
        time.sleep(30)

# ------------------ Hotkey registration on startup ------------------

def register_hotkeys():
    hot = brain.get("hotkeys", {})
    for keys, routine in hot.items():
        try:
            keyboard.add_hotkey(keys, lambda r=routine: run_routine(r))
        except Exception as e:
            print("Failed to bind hotkey", keys, e)

# ------------------ Simple Tkinter GUI ------------------

class JarvisGUI:
    def __init__(self, root):
        self.root = root
        root.title("Jarvis 5.0")
        root.geometry("520x420")
        # status label
        self.status_var = tk.StringVar(value="Idle")
        ttk.Label(root, text="Jarvis 5.0", font=(None,16)).pack(pady=6)
        ttk.Label(root, textvariable=self.status_var).pack()

        frame = ttk.Frame(root)
        frame.pack(pady=8, fill=tk.X, padx=8)

        ttk.Button(frame, text="Start Listening", command=self.start_listen).grid(row=0, column=0, padx=4, pady=4)
        ttk.Button(frame, text="Stop Listening", command=self.stop_listen).grid(row=0, column=1, padx=4, pady=4)
        ttk.Button(frame, text="Run Selected Routine", command=self.run_selected).grid(row=0, column=2, padx=4, pady=4)

        # routines list
        self.routine_list = tk.Listbox(root, height=8)
        self.routine_list.pack(fill=tk.X, padx=8)
        self.refresh_routines()

        btn_frame = ttk.Frame(root)
        btn_frame.pack(pady=6)
        ttk.Button(btn_frame, text="Create Routine (Voice)", command=self.create_routine_voice).grid(row=0, column=0, padx=6)
        ttk.Button(btn_frame, text="Schedule Routine (Voice)", command=self.schedule_routine_voice).grid(row=0, column=1, padx=6)
        ttk.Button(btn_frame, text="Bind Hotkey (Voice)", command=self.bind_hotkey_voice).grid(row=0, column=2, padx=6)

        # logs
        ttk.Label(root, text="Logs:").pack(anchor=tk.W, padx=8)
        self.log_box = tk.Text(root, height=8)
        self.log_box.pack(fill=tk.BOTH, padx=8, pady=4, expand=True)

        self.listening = False
        self.listener_thread = None

    def log(self, msg):
        ts = datetime.now().strftime("%H:%M:%S")
        self.log_box.insert(tk.END, f"[{ts}] {msg}\n")
        self.log_box.see(tk.END)

    def refresh_routines(self):
        self.routine_list.delete(0, tk.END)
        for name in brain.get("routines", {}).keys():
            self.routine_list.insert(tk.END, name)

    def start_listen(self):
        if self.listening:
            return
        self.listening = True
        self.status_var.set("Listening")
        self.listener_thread = threading.Thread(target=self.listen_loop, daemon=True)
        self.listener_thread.start()
        self.log("Started voice loop")

    def stop_listen(self):
        if not self.listening:
            return
        self.listening = False
        self.status_var.set("Idle")
        self.log("Stopped voice loop")

    def listen_loop(self):
        while self.listening:
            cmd = listen()
            if cmd:
                self.log(f"Heard: {cmd}")
                res = execute(cmd)
                if res == "exit":
                    self.listening = False
                    self.status_var.set("Idle")
                    break
            time.sleep(0.2)

    def run_selected(self):
        sel = self.routine_list.curselection()
        if not sel:
            messagebox.showinfo("Info","Select a routine first")
            return
        name = self.routine_list.get(sel[0])
        self.log(f"Running routine {name}")
        threading.Thread(target=run_routine, args=(name,), daemon=True).start()

    def create_routine_voice(self):
        speak("Say the name of the routine")
        name = listen()
        if not name:
            messagebox.showinfo("Info","No name detected")
            return
        success = create_routine_via_voice(name)
        if success:
            self.refresh_routines()
            self.log(f"Created routine {name}")

    def schedule_routine_voice(self):
        speak("Say schedule command like: schedule routine morning at 07:30 daily")
        cmd = listen(10,8)
        if cmd:
            ok = schedule_routine_via_voice(cmd)
            if ok:
                self.log(f"Scheduled: {cmd}")
            else:
                self.log("Schedule failed")

    def bind_hotkey_voice(self):
        speak("Say bind command like: bind routine morning to ctrl alt m")
        cmd = listen(8,6)
        if cmd:
            ok = bind_hotkey_via_voice(cmd)
            if ok:
                self.log(f"Bound: {cmd}")
            else:
                self.log("Bind failed")
